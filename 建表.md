-- ============================================================
-- 超市管理系统 (SupermarketManage) 数据库建表脚本
-- 数据库名: supermarket_manage
-- 字符集: utf8mb4  |  引擎: InnoDB
-- ============================================================

-- 1. 用户账户表
CREATE TABLE `user_info` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '用户ID',
  `username` varchar(50) NOT NULL COMMENT '用户名，用于登录',
  `user_type` tinyint NOT NULL DEFAULT '1' COMMENT '用户类型：0-管理员 1-普通用户',
  `password` varchar(255) NOT NULL COMMENT '密码，使用BCrypt算法加密存储',
  `phone` varchar(20) NOT NULL COMMENT '手机号，注销时追加-数字后缀实现复用',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `uk_phone` (`phone`) USING BTREE,
  UNIQUE KEY `uk_username` (`username`) USING BTREE,
  CONSTRAINT `ck_user_info_phone` CHECK (regexp_like(`phone`,_utf8mb4'^1[3-9][0-9]{9}(-[0-9]+)?$')),
  CONSTRAINT `ck_user_type` CHECK ((`user_type` in (0,1)))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='用户账户表';

-- 2. 短信验证码表
CREATE TABLE `sms_code` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '记录ID',
  `phone` varchar(11) NOT NULL COMMENT '11位手机号',
  `user_id` bigint unsigned DEFAULT NULL COMMENT '关联user_info.id',
  `code` varchar(6) NOT NULL COMMENT '6位数字验证码',
  `expire_time` datetime NOT NULL COMMENT '过期时间（发送后5分钟）',
  `is_used` tinyint NOT NULL DEFAULT '0' COMMENT '是否已使用：0-未使用 1-已使用',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '发送时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_phone_used` (`phone`,`is_used`) USING BTREE,
  KEY `fk_sms_code_user` (`user_id`) USING BTREE,
  CONSTRAINT `fk_sms_code_user` FOREIGN KEY (`user_id`) REFERENCES `user_info` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `ck_is_used` CHECK ((`is_used` in (0,1)))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='短信验证码表';

-- 3. 管理员注册校验码表
CREATE TABLE `admin_register_code` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '校验码ID',
  `code` varchar(32) NOT NULL COMMENT '管理员注册校验码',
  `total_usable_times` int unsigned NOT NULL DEFAULT '1' COMMENT '总可使用次数',
  `used_times` int unsigned NOT NULL DEFAULT '0' COMMENT '已使用次数',
  `is_valid` tinyint NOT NULL DEFAULT '1' COMMENT '是否有效：0-无效 1-有效',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `uk_code` (`code`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='管理员注册校验码表';

-- 4. 用户反馈表
CREATE TABLE `user_feedback` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '反馈ID',
  `user_id` bigint unsigned NOT NULL COMMENT '提交用户ID',
  `content` text NOT NULL COMMENT '反馈内容',
  `reply` text COMMENT '管理员回复',
  `status` tinyint NOT NULL DEFAULT '0' COMMENT '处理状态：0=未处理 1=已回复',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '提交时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_user_id` (`user_id`) USING BTREE,
  KEY `idx_status` (`status`) USING BTREE,
  CONSTRAINT `fk_feedback_user` FOREIGN KEY (`user_id`) REFERENCES `user_info` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='用户反馈表';

-- 5. 商品分类表
CREATE TABLE `product_category` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '分类ID',
  `category_name` varchar(50) NOT NULL COMMENT '分类名称',
  `sort` int NOT NULL DEFAULT '0' COMMENT '排序序号，数字越小越靠前',
  `is_enabled` tinyint NOT NULL DEFAULT '1' COMMENT '启用状态：0-禁用 1-启用',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `uk_category_name` (`category_name`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='商品分类表';

-- 6. 商品基础档案表
CREATE TABLE `product_base` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '商品基础ID',
  `category_id` bigint unsigned NOT NULL COMMENT '关联商品分类ID',
  `product_name` varchar(100) NOT NULL COMMENT '商品标准名称',
  `is_enabled` tinyint NOT NULL DEFAULT '1' COMMENT '启用状态：0-禁用 1-启用',
  `image_url` varchar(500) DEFAULT NULL COMMENT '商品图片URL',
  `crop_data` varchar(500) DEFAULT NULL COMMENT '图片裁剪定位数据JSON',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `fk_product_category` (`category_id`) USING BTREE,
  KEY `idx_product_name` (`product_name`) USING BTREE,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='商品基础登记表';

-- 7. 进货入库记录表
CREATE TABLE `purchase_record` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '进货记录ID',
  `base_product_id` bigint unsigned DEFAULT NULL COMMENT '关联商品基础档案ID',
  `purchase_quantity` int unsigned NOT NULL COMMENT '进货数量',
  `purchase_unit_price` decimal(10,2) NOT NULL COMMENT '进货单价（成本价）',
  `purchase_total_amount` decimal(10,2) NOT NULL COMMENT '进货总金额',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '入库时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_create_time` (`create_time`) USING BTREE,
  KEY `idx_purchase_base_product` (`base_product_id`),
  CONSTRAINT `ck_purchase_price_positive` CHECK ((`purchase_unit_price` >= 0)),
  CONSTRAINT `ck_purchase_quantity_positive` CHECK ((`purchase_quantity` > 0))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='进货入库记录表';

-- 8. 货架信息表
CREATE TABLE `shelf_info` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '货架ID',
  `shelf_code` varchar(30) NOT NULL COMMENT '货架编号，如A-01、B-03',
  `shelf_area` varchar(50) NOT NULL COMMENT '所属区域，如食品区、日化区',
  `shelf_layer` int DEFAULT NULL COMMENT '货架层数',
  `layer_capacity` int unsigned DEFAULT NULL COMMENT '每层货物容量（件）',
  `total_capacity` int unsigned DEFAULT NULL COMMENT '货架总容量 = 层数 × 每层容量',
  `is_enabled` tinyint NOT NULL DEFAULT '1' COMMENT '启用状态：0-禁用 1-启用',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `uk_shelf_code` (`shelf_code`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='货架信息表';

-- 9. 货架每层容量追踪表
CREATE TABLE `shelf_capacity_tracking` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '记录ID',
  `shelf_id` bigint unsigned NOT NULL COMMENT '关联货架ID',
  `shelf_layer` int NOT NULL COMMENT '层数编号，从1开始',
  `used_count` int unsigned NOT NULL DEFAULT '0' COMMENT '该层已使用数量',
  `max_capacity` int unsigned NOT NULL COMMENT '该层最大容量（同步自shelf_info.layer_capacity）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `uk_shelf_layer` (`shelf_id`,`shelf_layer`),
  CONSTRAINT `fk_capacity_shelf` FOREIGN KEY (`shelf_id`) REFERENCES `shelf_info` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='货架每层容量追踪表';

-- 10. 销售订单主表
CREATE TABLE `sale_order` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '订单ID',
  `user_id` bigint unsigned NOT NULL COMMENT '操作人ID（关联user_info）',
  `pay_amount` decimal(10,2) NOT NULL COMMENT '订单支付总金额',
  `pay_status` tinyint NOT NULL DEFAULT '0' COMMENT '支付状态：0-未支付 1-已支付 2-已退款(全部) 3-已取消 4-部分退款',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `fk_order_user` (`user_id`) USING BTREE,
  KEY `idx_pay_status` (`pay_status`) USING BTREE,
  KEY `idx_create_time` (`create_time`) USING BTREE,
  CONSTRAINT `fk_order_user` FOREIGN KEY (`user_id`) REFERENCES `user_info` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `ck_pay_amount_positive` CHECK ((`pay_amount` >= 0)),
  CONSTRAINT `ck_pay_status` CHECK ((`pay_status` in (0,1,2,3,4)))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='销售订单主表';

-- 11. 销售订单详情表
CREATE TABLE `sale_order_item` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '订单详情ID',
  `order_id` bigint unsigned NOT NULL COMMENT '关联订单ID',
  `base_product_id` bigint unsigned DEFAULT NULL COMMENT '关联商品基础档案ID',
  `batch_detail_id` bigint unsigned DEFAULT NULL COMMENT '关联批次明细ID，用于成本追溯',
  `buy_quantity` int unsigned NOT NULL COMMENT '购买数量',
  `buy_unit_price` decimal(10,2) NOT NULL COMMENT '成交单价',
  `item_total_amount` decimal(10,2) NOT NULL COMMENT '单项总金额',
  `refund_quantity` int unsigned NOT NULL DEFAULT '0' COMMENT '累计退货数量',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `fk_item_order` (`order_id`) USING BTREE,
  KEY `idx_batch_detail` (`batch_detail_id`),
  CONSTRAINT `fk_item_order` FOREIGN KEY (`order_id`) REFERENCES `sale_order` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ck_buy_quantity_positive` CHECK ((`buy_quantity` > 0)),
  CONSTRAINT `ck_refund_quantity_limit` CHECK ((`refund_quantity` <= `buy_quantity`))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='销售订单详情表';

-- 12. 商品批次明细表（库存核心，每件商品独立一条记录，全生命周期跟踪）
CREATE TABLE `product_batch_detail` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '批次明细ID',
  `purchase_record_id` bigint unsigned NOT NULL COMMENT '关联进货记录ID',
  `base_product_id` bigint unsigned NOT NULL COMMENT '关联商品基础档案ID',
  `shelf_id` bigint unsigned DEFAULT NULL COMMENT '关联货架ID，NULL=未上架',
  `shelf_layer` int DEFAULT NULL COMMENT '所在货架层数，NULL=未分配',
  `shelf_display_name` varchar(100) DEFAULT NULL COMMENT '货架展示名称（上架时填写，下架/售出/损耗时清空）',
  `purchase_unit_price` decimal(10,2) NOT NULL COMMENT '进货成本单价（写入后永久不变，用于成本核算）',
  `sale_price` decimal(10,2) NOT NULL DEFAULT '0.00' COMMENT '销售单价（上架时填写）',
  `discount_price` decimal(10,2) DEFAULT NULL COMMENT '折扣价格（上架时可选填写）',
  `status` tinyint NOT NULL DEFAULT '0' COMMENT '状态：0=在库未上架 1=已上架在售 2=已售出 3=已损耗 4=退货待处理 5=待支付锁定；退货回库存 2→1；退货转损耗 2→3',
  `belong_order_item_id` bigint unsigned DEFAULT NULL COMMENT '关联销售订单详情ID（售出/锁定时写入，解锁/退货回库时清空）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_base_product_id` (`base_product_id`) USING BTREE,
  KEY `idx_shelf_id` (`shelf_id`) USING BTREE,
  KEY `idx_status` (`status`) USING BTREE,
  KEY `idx_purchase_record_id` (`purchase_record_id`) USING BTREE,
  KEY `fk_batch_order_item` (`belong_order_item_id`),
  CONSTRAINT `fk_batch_order_item` FOREIGN KEY (`belong_order_item_id`) REFERENCES `sale_order_item` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `fk_batch_product` FOREIGN KEY (`base_product_id`) REFERENCES `product_base` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `fk_batch_purchase` FOREIGN KEY (`purchase_record_id`) REFERENCES `purchase_record` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `fk_batch_shelf` FOREIGN KEY (`shelf_id`) REFERENCES `shelf_info` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='商品批次明细表 - 库存核心，每件商品独立记录，全生命周期跟踪';

-- 13. 销售退货记录表
CREATE TABLE `sale_refund` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '退货记录ID',
  `order_id` bigint unsigned NOT NULL COMMENT '关联原订单ID',
  `order_item_id` bigint unsigned NOT NULL COMMENT '关联原订单详情ID',
  `batch_detail_id` bigint unsigned DEFAULT NULL COMMENT '关联批次明细ID',
  `operator_id` bigint unsigned NOT NULL COMMENT '退货操作人ID（收银员/管理员）',
  `refund_quantity` int unsigned NOT NULL COMMENT '本次退货数量',
  `refund_unit_price` decimal(10,2) NOT NULL COMMENT '退货单价（按原成交价退款）',
  `refund_total_amount` decimal(10,2) NOT NULL COMMENT '本次退款总金额',
  `refund_reason` varchar(255) DEFAULT NULL COMMENT '退货原因',
  `refund_handle_status` tinyint NOT NULL DEFAULT '0' COMMENT '处理状态：0=待审核 1=已同意（退款完成） 2=已拒绝',
  `back_stock_quantity` int unsigned DEFAULT '0' COMMENT '管理员审核填写的回库数量（退回库存恢复上架，批次状态 2→1）',
  `to_loss_quantity` int unsigned DEFAULT '0' COMMENT '管理员审核填写的转损耗数量（批次状态 2→3，同时生成loss_record记录）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '申请时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '处理时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `fk_refund_order` (`order_id`) USING BTREE,
  KEY `fk_refund_item` (`order_item_id`) USING BTREE,
  KEY `fk_refund_operator` (`operator_id`) USING BTREE,
  KEY `idx_create_time` (`create_time`) USING BTREE,
  KEY `idx_refund_handle_status` (`refund_handle_status`),
  CONSTRAINT `fk_refund_item` FOREIGN KEY (`order_item_id`) REFERENCES `sale_order_item` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `fk_refund_operator` FOREIGN KEY (`operator_id`) REFERENCES `user_info` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `fk_refund_order` FOREIGN KEY (`order_id`) REFERENCES `sale_order` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='销售退货记录表 - 记录每笔退货申请及审核结果';

-- 14. 商品损耗记录表
CREATE TABLE `loss_record` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '损耗记录ID',
  `batch_detail_id` bigint unsigned DEFAULT NULL COMMENT '关联批次明细ID，成本从对应批次进货单价取值',
  `loss_reason` varchar(255) NOT NULL COMMENT '损耗原因（破损、过期、丢失、退货转损耗等）',
  `loss_quantity` int unsigned NOT NULL COMMENT '损耗数量',
  `loss_unit_price` decimal(10,2) NOT NULL COMMENT '损耗单价（按成本价计算）',
  `loss_total_amount` decimal(10,2) NOT NULL COMMENT '损耗总金额',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '登记时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_create_time` (`create_time`) USING BTREE,
  KEY `idx_loss_batch` (`batch_detail_id`),
  CONSTRAINT `ck_loss_price_positive` CHECK ((`loss_unit_price` >= 0)),
  CONSTRAINT `ck_loss_quantity_positive` CHECK ((`loss_quantity` > 0))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='商品损耗记录表 - 登记直接损耗及退货转损耗';
